Plot differences

df_complete <- df_complete %>%
  mutate(
    w2v_error = Word2VecSimilarity - MeanSimilarity,
    minilm_error = MiniLMSimilarity - MeanSimilarity,
    mpnet_error = MPNetSimilarity - MeanSimilarity
  )

df_long <- df_complete %>%
  pivot_longer(
    cols = c(w2v_error, minilm_error, mpnet_error),
    names_to = "Model",
    values_to = "Error"
  )

ggplot(df_long,
       aes(x = FeatureMatch,
           y = Error,
           color = Model,
           group = Model)) +
  stat_summary(fun = mean, geom = "line", linewidth = 1.2) +
  stat_summary(fun = mean, geom = "point") +
  theme_classic() +
  labs(y = "Model − Human (Error)")
